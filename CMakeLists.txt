cmake_minimum_required(VERSION 3.20)
project(xphone VERSION 0.1.0 LANGUAGES C)
include(GNUInstallDirs)
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
option(XPHONE_SANITIZE "Enable address and undefined behavior sanitizers" OFF)
option(XPHONE_BUILD_CLIENT "Build the X11 client and audio tests" ON)
find_package(PkgConfig REQUIRED)
find_package(Threads REQUIRED)
pkg_check_modules(NET REQUIRED IMPORTED_TARGET libevent_openssl openssl>=3.5 libngtcp2>=1.22 libngtcp2_crypto_ossl libnghttp3>=1.14 libnghttp2 jansson)
if(XPHONE_BUILD_CLIENT)
  pkg_check_modules(DESKTOP REQUIRED IMPORTED_TARGET x11 xft fontconfig xi libpulse opus speexdsp)
endif()
add_compile_definitions(_GNU_SOURCE)
add_compile_options(-Wall -Wextra -Wpedantic -Wshadow -Wformat=2 -Wstrict-prototypes)
if(XPHONE_SANITIZE)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
endif()
add_library(xphone_net src/common.c src/wire.c src/transport.c src/http2.c src/http3.c)
target_include_directories(xphone_net PUBLIC src)
target_link_libraries(xphone_net PUBLIC PkgConfig::NET Threads::Threads m)
add_executable(xphoned src/server.c)
target_link_libraries(xphoned PRIVATE xphone_net)
if(XPHONE_BUILD_CLIENT)
  add_library(xphone_client src/session.c src/audio.c)
  target_link_libraries(xphone_client PUBLIC xphone_net PkgConfig::DESKTOP)
  add_executable(xphone src/client.c src/ui.c)
  target_link_libraries(xphone PRIVATE xphone_client)
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(test_wire tests/test_wire.c)
  target_compile_options(test_wire PRIVATE -UNDEBUG)
  target_link_libraries(test_wire PRIVATE xphone_net)
  add_test(NAME wire COMMAND test_wire)
  if(XPHONE_BUILD_CLIENT)
    add_executable(test_audio tests/test_audio.c)
    target_compile_options(test_audio PRIVATE -UNDEBUG)
    target_link_libraries(test_audio PRIVATE xphone_client)
    add_test(NAME audio COMMAND test_audio)
    find_package(Python3 COMPONENTS Interpreter REQUIRED)
    add_test(NAME integration COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/integration.py $<TARGET_FILE:xphoned> $<TARGET_FILE:xphone>)
    set_tests_properties(integration PROPERTIES TIMEOUT 120)
    find_program(XVFB_RUN xvfb-run)
    find_program(XDOTOOL xdotool)
    find_program(IMAGE_IMPORT import)
    if(XVFB_RUN AND XDOTOOL AND IMAGE_IMPORT)
      add_test(NAME gui COMMAND ${XVFB_RUN} -a -s "-screen 0 2048x1600x24" ${Python3_EXECUTABLE}
        ${CMAKE_SOURCE_DIR}/tests/gui.py $<TARGET_FILE:xphoned> $<TARGET_FILE:xphone> ${CMAKE_BINARY_DIR}/screenshots)
      set_tests_properties(gui PROPERTIES TIMEOUT 90)
    endif()
  endif()
endif()
install(TARGETS xphoned RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
if(XPHONE_BUILD_CLIENT)
  install(TARGETS xphone RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
  install(FILES packaging/xphone.desktop DESTINATION ${CMAKE_INSTALL_DATADIR}/applications)
endif()
install(FILES packaging/xphoned.service DESTINATION lib/systemd/system)
install(PROGRAMS packaging/xphoned.initd
  DESTINATION ${CMAKE_INSTALL_DATADIR}/xphone/openrc/init.d RENAME xphoned)
install(FILES packaging/xphoned.confd
  DESTINATION ${CMAKE_INSTALL_DATADIR}/xphone/openrc/conf.d RENAME xphoned)
install(FILES README.md docs/protocol.md docs/third-party.md DESTINATION ${CMAKE_INSTALL_DOCDIR})
install(DIRECTORY examples/ DESTINATION ${CMAKE_INSTALL_DOCDIR}/examples)
