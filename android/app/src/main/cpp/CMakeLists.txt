cmake_minimum_required(VERSION 3.20)
project(xphone_android LANGUAGES C)
set(CMAKE_C_STANDARD 17)
get_filename_component(XPHONE_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../../../.." ABSOLUTE)
set(SOURCES bridge.c ${XPHONE_ROOT}/src/common.c ${XPHONE_ROOT}/src/wire.c
    ${XPHONE_ROOT}/src/transport.c ${XPHONE_ROOT}/src/http2.c ${XPHONE_ROOT}/src/http3.c)
add_library(xphone_android SHARED ${SOURCES})
target_compile_definitions(xphone_android PRIVATE _GNU_SOURCE)
target_compile_options(xphone_android PRIVATE -Wall -Wextra -Werror -Wno-unused-parameter)
target_include_directories(xphone_android PRIVATE ${XPHONE_ROOT}/src)
if(ANDROID)
    set(PREFIX "${XPHONE_DEPS}/${ANDROID_ABI}")
    if(NOT EXISTS "${PREFIX}/lib/libngtcp2_crypto_ossl.a")
        message(FATAL_ERROR "Run android/tools/build-native-deps.py with the NDK first; missing ${PREFIX}")
    endif()
    target_include_directories(xphone_android PRIVATE ${PREFIX}/include)
    # All archives are PIC and built for this ABI by build-native-deps.py.
    foreach(LIB event_openssl event ngtcp2_crypto_ossl ngtcp2 nghttp3 nghttp2 jansson opus ssl crypto)
        target_link_libraries(xphone_android PRIVATE "${PREFIX}/lib/lib${LIB}.a")
    endforeach()
    target_link_libraries(xphone_android PRIVATE m dl)
    target_link_options(xphone_android PRIVATE -Wl,-z,max-page-size=16384)
else()
    # Host JNI build exercises the exact bridge against the real server without an emulator.
    find_package(JNI REQUIRED)
    find_package(PkgConfig REQUIRED)
    pkg_check_modules(DEPS REQUIRED IMPORTED_TARGET libevent_openssl openssl>=3.5
        libngtcp2>=1.22 libngtcp2_crypto_ossl libnghttp3>=1.14 libnghttp2 jansson opus)
    target_include_directories(xphone_android PRIVATE ${JNI_INCLUDE_DIRS})
    target_link_libraries(xphone_android PRIVATE PkgConfig::DEPS m)
endif()
